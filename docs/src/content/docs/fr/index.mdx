import { Card, CardGrid } from '@astrojs/starlight/components';
import demo from '../../../assets/demo.gif';

<img class="demo" src={demo.src} alt="Notosaurus sur un téléphone : une page de vocabulaire devient des cartes, un schéma voit ses légendes masquées, des cartes de géométrie reçoivent des figures exactes" />

<CardGrid stagger>
	<Card title="À partir d'une photo ou d'un PDF" icon="seti:image">
		Prenez en photo une page de cahier ou de manuel, ou ajoutez un PDF. Une IA la lit, écriture
		manuscrite comprise, et prépare les cartes.
	</Card>
	<Card title="Vous gardez la main" icon="pencil">
		Relisez chaque carte, ou demandez simplement : « ajoute le pluriel au verso », « tu as
		oublié les couleurs ».
	</Card>
	<Card title="Schémas, formules, figures" icon="puzzle">
		Les légendes d'un schéma cachées derrière des numéros, les formules en MathJax, des
		figures de géométrie exactes, des textes à trous et des QCM.
	</Card>
	<Card title="Directement dans Anki" icon="rocket">
		Un geste envoie le paquet dans Anki, avec l'audio. Révision sur AnkiDroid ou AnkiMobile,
		synchronisée par AnkiWeb.
	</Card>
	<Card title="Pensé pour les familles" icon="star">
		Une fiche par enfant : sa classe, ses paquets, ses propres instructions, et ses leçons.
		Son prénom ne va jamais à l'IA.
	</Card>
	<Card title="Sur le téléphone ou l'ordinateur" icon="github">
		L'application Android seule, avec un abonnement : ni ordinateur ni clé d'IA. Ou le
		greffon Anki libre et gratuit, avec votre propre clé d'IA (Gemini, Claude, OpenAI,
		OpenRouter, ou un modèle chez vous).
	</Card>
</CardGrid>
